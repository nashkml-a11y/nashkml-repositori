/* ---- renderPokedex ---- */
function renderPokedex(){
    let box=document.getElementById("pokedexList");
    if(!box) return;

    // Mascotas de temporada
    function getPetSeason(emoji){
      for(const key in SEASONS){
        if(SEASONS[key].pets && SEASONS[key].pets.includes(emoji)) return SEASONS[key];
      }
      return null;
    }

    let entries=Object.entries(pets);

    box.innerHTML=entries.map(([emoji,p])=>{
        let ownedKeys=Object.keys(game.pets||{}).filter(k=>k.startsWith(emoji+"|||") && (game.pets[k]||0)>0);
        let currentlyOwned=ownedKeys.length>0;
        let discovered=!!(game.petsDiscovered && game.petsDiscovered[emoji]) || currentlyOwned;
        let shinyCount=Object.entries(game.shinyPets||{}).filter(([k,v])=>k.includes("|||"+emoji+"|||")&&v>0).reduce((a,[k,v])=>a+v,0);
        let highestLevel=currentlyOwned
            ? Math.max(...ownedKeys.map(k=>Number(k.split("|||")[1]||0)+1))
            : 0;

        const petSeason = getPetSeason(emoji);
        const seasonLocked = petSeason && !isSeasonActive(petSeason.id) && !discovered;

        if(seasonLocked){
          return `
          <div class="pokedexCard missing">
            <div class="pokedexPet missing">${emoji}</div>
            <h3>${p[0]}</h3>
            <div class="badge">${petSeason.emoji} ${petSeason.name}</div>
            <div class="small" style="margin-top:8px;color:#ffb84d">
              🔒 Exclusivo de la actualización de ${petSeason.name}<br>
              Disponible durante su mes en el Mundo de ${petSeason.name}
            </div>
          </div>`;
        }

        return `
        <div class="pokedexCard ${discovered?"":"missing"}">
            <div class="pokedexPet ${discovered?"":"missing"}">${emoji}</div>
            ${discovered
                ? `
                    <h3>${p[0]}</h3>
                    <div class="badge ${p[3]}">${p[1]}</div>
                    ${petSeason ? `<div class="small" style="color:#ff9a3c">${petSeason.emoji} ${petSeason.name}</div>` : ""}
                    <div class="multiplier ${p[3]}">⚡ x${Number(p[2]).toFixed(2)}</div>
                    <div class="small">Nivel máximo: <b>${highestLevel}</b></div>
                    <div class="small" style="margin-top:5px">✅ Descubierta ${shinyCount?`· ✨ Shiny x${shinyCount}`:""}${currentlyOwned?"":" · (sin copias actuales)"}</div>
                  `

/* ---- renderTime ---- */
function renderTime(){

let x=
document.getElementById("time");


if(x){

x.textContent=
formatTime();

}


let box=
document.getElementById(
"timeRewards"
);


if(!box){
    return;
}


let items=
box.querySelectorAll(".item");


/* Primera construcción, o cambió el número de recompensas:
   reconstruimos todo desde cero */
if(items.length!==timeRewards.length){

box.innerHTML=
timeRewards
.map((r,i)=>timeRewardItemHTML(i,r))
.join("");

return;

}


/* Ya existen las tarjetas: actualizamos SOLO lo que cambia
   (ancho de la barra, texto del botón) sin recrear los nodos,
   para que la animación de la barra no se reinicie cada segundo */
timeRewards.forEach((r,i)=>{

let itemEl=items[i];

let progress=
Math.min(100,game.playSeconds/r[0]*100);

let bar=
itemEl.querySelector(".bar");

if(bar){
    bar.style.width=progress+"%";
}

let claimed=
game.timeClaimed[i];

let unlocked=
game.playSeconds>=r[0];

let state=
claimed?"claimed":(unlocked?"claimable":"locked");

let btn=
itemEl.querySelector("button");

if(itemEl.dataset.state!==state){

    /* Cambio real de estado (se desbloqueó o se reclamó):
       sustituimos solo el botón de esta tarjeta */
    if(btn){
        btn.outerHTML=
        timeRewardButtonHTML(i,r,claimed,unlocked,progress);
    }

    itemEl.dataset.state=state;

}else if(state==="locked" && btn){

    /* Solo cambia el porcentaje: actualizamos el texto
       sin recrear el botón */
    btn.textContent=
    "🔒 "+Math.floor(progress)+"%";

}

});

}

            p.style.width="";

            p.style.background="";

            p.style.marginTop=(-size/2)+"px";
            p.style.background=colors[Math.floor(Math.random()*colors.length)];
        }

        const angle=(Math.PI*2/count)*i+Math.random()*0.5;
        const dist=60+Math.random()*50;
        p.style.setProperty("--pfly","translate("+(Math.cos(angle)*dist)+"px,"+(Math.sin(angle)*dist)+"px)");

            cl1RestartAnimation(p,"particleFly .7s ease-out forwards");

    }

    /* Resorte real: conserva la transformación base de cada skin y añade
       squash/stretch encima. Si el dispositivo no soporta Web Animations,
       el estado visual de pulsación sigue funcionando como respaldo. */
    try{
        if(btn._springAnimation) btn._springAnimation.cancel();
        const base=getComputedStyle(btn).transform;
        const t=(base && base!=="none") ? base+" " : "";
        btn._springAnimation=btn.animate([
            {transform:t+"scale(1,1)"},
            {transform:t+"scale(1.10,.90)"},
            {transform:t+"scale(.94,1.07)"},
            {transform:t+"scale(1.035,.975)"},
            {transform:t+"scale(1,1)"}
        ],{duration:380,easing:"cubic-bezier(.2,.8,.2,1)"});
    }catch(e){}

    clearTimeout(btn._pressTimer);
    btn._pressTimer=setTimeout(()=>{
        btn.classList.remove("touchActive");
        btn.classList.remove("clickPressed");
    },120);
}

/* ---- tone ---- */
function tone(freq,dur,type,vol,delay){

    let ctx=getAudioCtx();
    if(!ctx) return;

    let t0=ctx.currentTime+(delay||0);

    let osc=ctx.createOscillator();
    let gain=ctx.createGain();

    osc.type=type||"sine";
    osc.frequency.setValueAtTime(freq,t0);

    gain.gain.setValueAtTime(0,t0);
    gain.gain.linearRampToValueAtTime(vol||0.15,t0+0.02);
    gain.gain.exponentialRampToValueAtTime(0.0001,t0+dur);

    osc.connect(gain);
    gain.connect(ctx.destination);

    osc.start(t0);
    osc.stop(t0+dur+0.05);

}

/* ---- chord ---- */
function chord(freqs,dur,type,vol,delay){
    freqs.forEach(f=>tone(f,dur,type,vol,delay));
}

/* ---- punch ---- */
function punch(freq,dur,delay){
    tone(freq,dur,"square",0.11,delay||0);
    tone(freq*0.5,dur,"sine",0.13,delay||0);
}

/* ---- sfx ---- */
function sfx(name){

    if(!soundOn()) return;

    switch(name){

        /* Clic crítico: chispazo agudo (rápido a propósito,
           se dispara muy a menudo) */
        case "crit":
            tone(880,0.12,"triangle",0.14,0);
            tone(1320,0.12,"triangle",0.10,0.05);
            break;

        /* Fusionar mascotas: brillo mágico ascendente + acorde final */
        case "fuse":
            tone(520,0.14,"sine",0.14,0);
            tone(780,0.14,"sine",0.14,0.09);
            tone(1040,0.16,"sine",0.14,0.18);
            chord([780,1040,1300],0.4,"triangle",0.11,0.28);
            break;

        /* Revelar mascota de un huevo: mini fanfarria juguetona */
        case "egg":
            tone(440,0.12,"square",0.10,0);
            tone(660,0.12,"square",0.10,0.07);
            chord([660,880,1100],0.35,"sawtooth",0.11,0.16);
            break;

        /* Reclamar misión / racha / recompensa por tiempo:
           fanfarria corta de "victoria" con acorde final */
        case "claim":
            tone(523.25,0.13,"triangle",0.16,0);
            tone(659.25,0.13,"triangle",0.16,0.1);
            tone(783.99,0.16,"triangle",0.17,0.2);
            chord([1046.5,1318.51,1567.98],0.45,"sawtooth",0.13,0.33);
            break;

        /* Desbloquear un mundo nuevo: fanfarria de trompetas
           (ondas de sierra = timbre más "metálico"/brillante)
           con golpe grave de cierre */
        case "unlock":
            tone(392,0.14,"sawtooth",0.13,0);
            tone(523.25,0.14,"sawtooth",0.13,0.12);
            tone(659.25,0.16,"sawtooth",0.14,0.24);
            chord([783.99,987.77,1174.66],0.55,"sawtooth",0.14,0.38);
            punch(196,0.4,0.38);
            break;

        /* Renacer: la más épica — swell grave, arpegio ascendente
           y acorde mayor completo con golpe final */
        case "rebirth":
            tone(130.81,0.5,"sawtooth",0.09,0);
            tone(261.63,0.16,"triangle",0.13,0.2);
            tone(329.63,0.16,"triangle",0.13,0.33);
            tone(392,0.16,"triangle",0.13,0.46);
            tone(523.25,0.2,"triangle",0.15,0.59);
            chord([523.25,659.25,783.99,1046.5],0.7,"sawtooth",0.13,0.76);
            punch(130.81,0.5,0.76);
            break;

        /* Canjear un código: chispa doble con brillo */
        case "code":
            tone(700,0.1,"triangle",0.14,0);
            tone(1050,0.16,"triangle",0.14,0.08);
            chord([1050,1400],0.3,"sawtooth",0.1,0.16);
            break;

        default:
            tone(600,0.1,"sine",0.12,0);

    }

}

/* ---- deletePetUnit ---- */
function deletePetUnit(k){
    if(!game.pets[k] || game.pets[k]<=0) return;
    if(petLocked(k)){
        toast("🔒 Esta mascota está protegida");
        return;
    }
    if(game.equipped.includes(k)){
        toast("❌ Primero quita esta mascota del equipo");
        return;
    }
    const reward = deletePetReward(k);
    if(!confirm("¿Eliminar 1 copia de esta mascota?\nRecibirás ~"+fmt(reward)+" 👆 del mundo actual.")) return;
    game.pets[k]--;
    if(game.pets[k]<=0){
        delete game.pets[k];
        delete game.petLocks[k];
        delete game.petFavorites[k];
    }
    W().clicks += reward;
    game.totalClicks += reward;
    toast("🗑️ Mascota eliminada · +"+fmt(reward)+" 👆");
    save(); render();
}
